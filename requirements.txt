# Runtime dependencies.
# Canonical source of truth — pyproject.toml reads this file, and the
# Dockerfiles and CI install from it. Two lists of pins drift.

# ---- Data ----
# pandas is pinned below 3.0 deliberately. Unpinned it resolves to 3.0.x, which
# changes the default string dtype from `object` to `str` and makes
# Copy-on-Write permanent. In a categorical-heavy regression pipeline that is a
# silent behaviour change, not an error — the worst kind. Revisit as a
# deliberate migration, not as a side effect of an install.
pandas>=2.2,<3
numpy>=2.0
pyarrow>=17.0
duckdb>=1.1

# ---- Ingestion ----
# `requests` only. Every source in the pipeline serves CSV or JSON, so no HTML
# parser is installed: beautifulsoup4 and lxml were carried for a Transfermarkt
# scrape that the licence forbids and an FBref one that Phase 12 declined
# (plans/02-fbref-spike.md). A parser with nothing to parse is a supply-chain
# surface with no upside.
requests>=2.32

# ---- Machine learning ----
scikit-learn>=1.9
xgboost>=3.0
lightgbm>=4.5
catboost>=1.2.7

# ---- Explainability ----
shap>=0.48

# ---- Visualisation ----
# matplotlib only. The interactive charts are Plotly, but they render in the
# browser from the npm package; the Python side ships static PNGs and imports
# plotly nowhere. It stays installed regardless — catboost requires it — but a
# direct pin here would claim this project chose it, and it did not.
matplotlib>=3.9

# ---- API ----
fastapi>=0.115
uvicorn[standard]>=0.32
pydantic>=2.9

# ---- Configuration ----
python-dotenv>=1.0
PyYAML>=6.0
